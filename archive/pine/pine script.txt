//@version=5
indicator("Rahul Momentum Options Engine (v4.5 trend/scalp RR + time + modes)", overlay = true, max_labels_count = 500)

//────────────────────────────────────
// CORE INDICATORS
//────────────────────────────────────
vwapVal = ta.vwap(close)
emaFast = ta.ema(close, 9)
emaSlow = ta.ema(close, 21)

volMA  = ta.sma(volume, 20)
body   = math.abs(close - open)
bodyMA = ta.sma(body, 20)

// Vol/impulse filters
volPctLookback   = input.int(100, "Volume percentile lookback", minval = 20)
volPctThreshold  = input.float(75.0, "Volume percentile threshold", minval = 1.0, maxval = 99.0)
bodyAtrMult      = input.float(0.5, "Min body vs ATR (impulse filter)", minval = 0.0, step = 0.05)

atr14  = ta.atr(14)
rsi14  = ta.rsi(close, 14)

// Distance from VWAP in ATR units
distFromVWAPATR       = math.abs(close - vwapVal) / atr14
nearVWAPEntry         = distFromVWAPATR <= 0.6
farFromVWAPEntry      = distFromVWAPATR >= 1.0
extendedFromVWAPEntry = distFromVWAPATR > 1.2  // very extended, fade risk

//────────────────────────────────────
// VISUAL / BEHAVIOR INPUTS
//────────────────────────────────────
entryLabelMult  = input.float(1.2, "Entry label distance (ATR)", minval = 0.5, step = 0.1)
exitLabelMult   = input.float(1.8, "Exit label distance (ATR)",  minval = 0.5, step = 0.1)
bannerAtrMult   = input.float(2.5, "Banner distance (ATR)",      minval = 1.0, step = 0.1)

showTinyMarks   = input.bool(true,  "Show tiny markers on candles")
useLunchFilter  = input.bool(true,  "Use no-trade zone (lunch)")
showRegimeBg    = input.bool(false, "Show regime background")

htfTF        = input.timeframe("5", "HTF timeframe for regime")
dxySymbol    = input.symbol("DXY",   "Dollar index symbol (DXY)")
us10ySymbol  = input.symbol("US10Y", "10Y yield symbol")
LunchSession = input.session("1130-1300", "No-trade session (exchange time)")
AfternoonTrendSession = input.session("1300-1600", "Trend re-arm session (exchange time)")

maxBarsInTrade = input.int(15, "Max bars in trade (time stop)", minval = 1, maxval = 100)
scalpTimeStopBars = input.int(12, "Max bars in trade (SCALP hard stop)", minval = 1, maxval = 100)
useWickRejectionExit = input.bool(true, "Use wick-rejection exit (stricter)")
minStructVwapAtrDist = input.float(0.6, "Min VWAP distance for structure (ATR)", minval = 0.0, step = 0.05)

// Stop / target config for lines
riskATRMult = input.float(1.0, "Base stop distance (ATR multiples)", minval = 0.1, step = 0.1)
rrMult      = input.float(2.0, "Target R multiple (TREND)",          minval = 0.5, step = 0.5)

// Separate RR for scalp / grind environments (defaults keep behavior the same)
rrMultTrend = rrMult
rrMultScalp = input.float(1.0, "Target R multiple (SCALP / grind)",  minval = 0.1, step = 0.1)

// Minimum expected move (in $ of underlying) required to take a SCALP entry
minScalpEdgeDollar = input.float(2.5, "Min expected move for scalps ($)", minval = 0.0, step = 0.1)

// Dynamic $-R + contract helper
acctEquity      = input.float(2000, "Account equity ($)", minval = 100)
riskPctPerTrade = input.float(1.0,  "Risk per trade (% of equity)", minval = 0.1, maxval = 5.0)
showSizeHelper  = input.bool(true,  "Show 1R $ & contract size guide")

// Micro-chop filter & vol-adjust
useChopFilter = input.bool(true,  "Use micro-chop filter")
useVolAdjust  = input.bool(true,  "Auto-adjust risk by volatility")

// VWAP mean-reversion scalps (aggressive, hints only)
enableScalpHints = input.bool(false, "Show VWAP mean-reversion scalp hints (aggressive)")

// Stats dashboard
statsLookback  = input.int(20, "Stats lookback (trades)", minval = 5, maxval = 100)
showStatsTable = input.bool(true, "Show mini stats dashboard")

//────────────────────────────────────
// MOMENTUM EXTENSION LOGIC (CALL / PUT)
//────────────────────────────────────

// VWAP slope
vwapUp   = vwapVal > vwapVal[1]
vwapDown = vwapVal < vwapVal[1]

// HTF trend (on same symbol) – computed early for momentum alignment and regime use
htfClose  = request.security(syminfo.tickerid, htfTF, close)
htfEMA20  = request.security(syminfo.tickerid, htfTF, ta.ema(close, 20))
htfUp     = htfClose > htfEMA20
htfDown   = htfClose < htfEMA20

// Micro-chop: price hugging VWAP & EMAs tight
emaSpread   = math.abs(emaFast - emaSlow)
microChop   = distFromVWAPATR < 0.2 and emaSpread < atr14 * 0.05
allowTrade  = not useChopFilter or not microChop

// Vol/impulse qualifiers (use percent rank to avoid unsupported percentile functions)
volRank     = ta.percentrank(volume, volPctLookback)
volStrong   = volRank >= volPctThreshold
bodyImpulse = body >= atr14 * bodyAtrMult

// Bar quality (close location in range)
barRange        = high - low
barQualityLong  = barRange > 0 ? close >= (low + barRange * 0.6) : false
barQualityShort = barRange > 0 ? close <= (high - barRange * 0.6) : false

// Wick filter to avoid heavy rejection bars
upperWick = high - math.max(open, close)
lowerWick = math.min(open, close) - low
wickOKLong  = barRange > 0 ? upperWick <= barRange * 0.5 : false
wickOKShort = barRange > 0 ? lowerWick <= barRange * 0.5 : false

// VWAP stretch guard for fresh entries (momentum path)
stretchBlockLong  = distFromVWAPATR > 1.2
stretchBlockShort = distFromVWAPATR > 1.2

// Bull momentum (CALL)
bullMomentum = close > vwapVal and emaFast > emaSlow and vwapUp and htfUp and body > bodyMA and bodyImpulse and volume > volMA and volStrong and barQualityLong and wickOKLong and allowTrade and not stretchBlockLong

// Bear momentum (PUT)
bearMomentum = close < vwapVal and emaFast < emaSlow and vwapDown and htfDown and body > bodyMA and bodyImpulse and volume > volMA and volStrong and barQualityShort and wickOKShort and allowTrade and not stretchBlockShort

//────────────────────────────────────
// REGIME DETECTION: TREND VS GRIND
//────────────────────────────────────

// RSI structure
rsiDirectional = rsi14 >= 60 or rsi14 <= 40
rsiCenterish   = rsi14 > 45 and rsi14 < 55

isTrendRegime   = farFromVWAPEntry and rsiDirectional and (htfUp or htfDown)
isGrindRegime   = nearVWAPEntry and rsiCenterish and not isTrendRegime
isNeutralRegime = not isTrendRegime and not isGrindRegime

// NEW: slow grind trend / bleeder day flag
// Trend is active but price is not very far from VWAP and EMAs are tight
isBleedTrend = isTrendRegime and distFromVWAPATR < 0.8 and emaSpread < atr14 * 0.15

// Trade mode flag: grind / bleed trend entries are treated as SCALP mode
bool isScalpTrade = isBleedTrend

// Mode classification
bool isTrendMode = isTrendRegime and not isBleedTrend
bool isScalpMode = not isTrendMode

// Optional background shading
regimeColor =
     isTrendRegime   ? color.new(color.green,  85) :
     isGrindRegime   ? color.new(color.orange, 88) :
     isNeutralRegime ? color.new(color.gray,   90) :
                       na

bgcolor(showRegimeBg ? regimeColor : na)

//────────────────────────────────────
// MACRO WIND (DXY + 10Y)
//────────────────────────────────────
dxyClose = request.security(dxySymbol, "60", close)
dxyMA20  = request.security(dxySymbol, "60", ta.sma(close, 20))
usdUp    = dxyClose > dxyMA20

us10yClose = request.security(us10ySymbol, "60", close)
us10yMA20  = request.security(us10ySymbol, "60", ta.sma(close, 20))
yieldsUp   = us10yClose > us10yMA20

macroBullMetals = (not usdUp) and (not yieldsUp)        // good for GLD/SLV
macroBearMetals = usdUp and yieldsUp                    // headwind
macroNeutral    = not macroBullMetals and not macroBearMetals

//────────────────────────────────────
// VOLATILITY REGIME & RISK ADJUST
//────────────────────────────────────
dayATR   = request.security(syminfo.tickerid, "D", ta.atr(14))
dayClose = request.security(syminfo.tickerid, "D", close)
volPct   = dayClose > 0 ? (dayATR / dayClose) * 100.0 : 0.0

volHigh = volPct > 3.0
volLow  = volPct < 1.5

volAdjMult =
     not useVolAdjust ? 1.0 :
     volHigh          ? 0.8 :   // tighter risk when vol high
     volLow           ? 1.2 :   // slightly wider risk when vol low
                           1.0

effRiskATRMult = riskATRMult * volAdjMult

// Expected move in underlying $ based on current ATR, mode, and RR
float expectedMoveDollar = atr14 * effRiskATRMult * (isTrendMode ? rrMultTrend : rrMultScalp)

// Require a minimum edge only in scalp mode; trend mode always allowed
bool scalpEdgeOK = not isScalpMode or expectedMoveDollar >= minScalpEdgeDollar

//────────────────────────────────────
// POSITION & CAMPAIGN STATE
//────────────────────────────────────
var bool inCall = false
var bool inPut  = false

// Campaign state
var float dayPnL_R    = 0.0

// Idea / campaign phases
var int ideaState        = 0
int     ideaStateProbe     = 0
int     ideaStateCore      = 1
int     ideaStateExpansion = 2
int     ideaStateOff       = -1

// Last-trade info for R-multiples
var float lastEntryPrice = na
var float lastRiskATR    = na   // 1R distance in price (already ATR-scaled)
var int   lastDirection  = 0    // +1 = CALL, -1 = PUT

// Per-trade progression helpers
var bool tp1Triggered      = false
var bool tp2Triggered      = false
var bool timeStopTriggered = false
var int  barsSinceEntry    = na

// Stop/target lines for *current* trade
var line  stopLine        = na
var line  targetLine      = na
var float currStopPrice   = na
var float currTargetPrice = na

// Stats: store last N trade R's
var float[] tradeRs = array.new_float()

// Dynamic dollar risk per trade
perTradeRiskDollar = acctEquity * riskPctPerTrade / 100.0

// Reset state at new daily session
if ta.change(time("D")) != 0
    inCall         := false
    inPut          := false
    dayPnL_R       := 0.0
    ideaState      := ideaStateProbe
    lastEntryPrice := na
    lastRiskATR    := na
    lastDirection  := 0
    tp1Triggered   := false
    tp2Triggered   := false
    timeStopTriggered := false
    barsSinceEntry := na

    // Do NOT delete lines here – keep history.
    stopLine        := na
    targetLine      := na
    currStopPrice   := na
    currTargetPrice := na

// Hard no-trade zone (e.g., lunch)
inLunchSession = time(timeframe.period, LunchSession) != 0
noTradeZone    = useLunchFilter and inLunchSession

// Afternoon re-arm session (for Trend → Neutral → Trend)
inAfternoonTrendSession = time(timeframe.period, AfternoonTrendSession) != 0

//────────────────────────────────────
// CAMPAIGN GUARDRAILS (no trade-count cap)
//────────────────────────────────────
canTradeToday = (ideaState != ideaStateOff)

//────────────────────────────────────
// SWING STRUCTURE (HH/HL/LH/LL) + REVERSAL FLAGS
//────────────────────────────────────
swingHigh = ta.pivothigh(high, 2, 2)
swingLow  = ta.pivotlow(low, 2, 2)

var float lastSwingHigh = na
var float prevSwingHigh = na
var float lastSwingLow  = na
var float prevSwingLow  = na

if not na(swingHigh)
    prevSwingHigh := lastSwingHigh
    lastSwingHigh := swingHigh

if not na(swingLow)
    prevSwingLow := lastSwingLow
    lastSwingLow := swingLow

isHH = not na(lastSwingHigh) and not na(prevSwingHigh) and lastSwingHigh > prevSwingHigh
isLH = not na(lastSwingHigh) and not na(prevSwingHigh) and lastSwingHigh < prevSwingHigh
isHL = not na(lastSwingLow)  and not na(prevSwingLow)  and lastSwingLow  > prevSwingLow
isLL = not na(lastSwingLow)  and not na(prevSwingLow)  and lastSwingLow  < prevSwingLow

// Require a minimum swing span (ATR-scaled) to qualify structure
minSwingAtrMult = input.float(1.5, "Min swing span (ATR multiple)", minval = 0.1, step = 0.1)
hasSpan = not na(lastSwingHigh) and not na(lastSwingLow) and math.abs(lastSwingHigh - lastSwingLow) >= (atr14 * minSwingAtrMult)

bullStruct = hasSpan and isHH and isHL
bearStruct = hasSpan and isLL and isLH

var int structState      = 0  // 0 none, 1 bull, -1 bear
int     structStateBull  = 1
int     structStateBear  = -1

bool bullReversalEvent = false
bool bearReversalEvent = false

int newStructState = structState
if bullStruct
    newStructState := structStateBull
else if bearStruct
    newStructState := structStateBear

if newStructState != structState
    bullReversalEvent := newStructState == structStateBull and structState == structStateBear
    bearReversalEvent := newStructState == structStateBear and structState == structStateBull
    structState := newStructState

//────────────────────────────────────
// ENTRY & EXIT CONDITIONS (EXECUTION)
//────────────────────────────────────
// Block first few minutes after RTH open on momentum path
blockOpenSession = time(timeframe.period, "0930-0933") != 0

buyCall = bullMomentum and not inCall and not inPut and not noTradeZone and not blockOpenSession and canTradeToday and scalpEdgeOK
buyPut  = bearMomentum and not inPut  and not inCall and not noTradeZone and not blockOpenSession and canTradeToday and scalpEdgeOK

// Structure-based entries still bypass lunch/chop filters but enforce vol/body, expected-move, and VWAP distance
structGuardOK = (ideaState != ideaStateOff)
structBuyCall = structGuardOK and bullStruct and close > vwapVal and emaFast > emaSlow and barQualityLong and wickOKLong and volStrong and bodyImpulse and scalpEdgeOK and distFromVWAPATR >= minStructVwapAtrDist and not inCall and not inPut
structBuyPut  = structGuardOK and bearStruct and close < vwapVal and emaFast < emaSlow and barQualityShort and wickOKShort and volStrong and bodyImpulse and scalpEdgeOK and distFromVWAPATR >= minStructVwapAtrDist and not inCall and not inPut

reversalBuyCall = structGuardOK and bullReversalEvent and hasSpan and close > vwapVal and emaFast > emaSlow and barQualityLong and wickOKLong and volStrong and bodyImpulse and scalpEdgeOK and distFromVWAPATR >= minStructVwapAtrDist and not inCall and not inPut
reversalBuyPut  = structGuardOK and bearReversalEvent and hasSpan and close < vwapVal and emaFast < emaSlow and barQualityShort and wickOKShort and volStrong and bodyImpulse and scalpEdgeOK and distFromVWAPATR >= minStructVwapAtrDist and not inCall and not inPut

buyCall := buyCall or structBuyCall or reversalBuyCall
buyPut  := buyPut  or structBuyPut  or reversalBuyPut

// MICRO STRUCTURE: pivot highs / lows for early exits
pivotHigh = high[1] > high[2] and high[1] > high
pivotLow  = low[1]  < low[2]  and low[1]  < low

earlySellCallStruct = inCall and pivotHigh and close < emaFast and close < close[1]
earlySellPutStruct  = inPut  and pivotLow  and close > emaFast and close > close[1]

// Structure fail exits
structFailSellCall = inCall and isLL and close < vwapVal
structFailSellPut  = inPut  and isHH and close > vwapVal

// Wick rejection exit (optional, stricter to avoid premature flattening)
wickRejectionCall = useWickRejectionExit and inCall and barRange > 0 and close <= (low + barRange * 0.25) and upperWick >= barRange * 0.6
wickRejectionPut  = useWickRejectionExit and inPut  and barRange > 0 and close >= (high - barRange * 0.25) and lowerWick >= barRange * 0.6

// Scalp hard time stop (independent of EMA cross)
hardTimeStopCall = inCall and isScalpTrade and not na(barsSinceEntry) and barsSinceEntry >= scalpTimeStopBars
hardTimeStopPut  = inPut  and isScalpTrade and not na(barsSinceEntry) and barsSinceEntry >= scalpTimeStopBars

// Original exit logic plus early structure-based exits
sellCall = (inCall and emaFast < emaSlow) or earlySellCallStruct or structFailSellCall or wickRejectionCall or hardTimeStopCall
sellPut  = (inPut  and emaFast > emaSlow) or earlySellPutStruct or structFailSellPut or wickRejectionPut or hardTimeStopPut

//────────────────────────────────────
// STATE MANAGEMENT (position flags)
//────────────────────────────────────
if buyCall
    inCall := true
    inPut  := false

if buyPut
    inPut  := true
    inCall := false

if sellCall
    inCall := false

if sellPut
    inPut := false

//────────────────────────────────────
// PRE-SIGNAL (EARLY WARNING) LOGIC
//────────────────────────────────────
preBuyCall =
     not inCall and not inPut and
     close > vwapVal and
     emaFast > emaSlow and
     not noTradeZone

preBuyPut =
     not inCall and not inPut and
     close < vwapVal and
     emaFast < emaSlow and
     not noTradeZone

//────────────────────────────────────
// CONFIDENCE METER (1–5)
//────────────────────────────────────
var float confidence = 0.0

confidence := 0.0
confidence += body   > bodyMA                           ? 1.0 : 0.0
confidence += volume > volMA                            ? 1.0 : 0.0
confidence += math.abs(emaFast - emaSlow) > atr14 * 0.2 ? 1.0 : 0.0
confidence += (vwapUp or vwapDown)                      ? 1.0 : 0.0
confidence += (rsi14 > 55 or rsi14 < 45)                ? 1.0 : 0.0
confidence := math.min(confidence, 5.0)

// Simple day bias view (for your headspace)
int dayBias = 0
dayBias :=
     emaFast > emaSlow and close > vwapVal ? 1 :
     emaFast < emaSlow and close < vwapVal ? -1 :
                                             0

//────────────────────────────────────
// TOP BANNER: DAY STATE + REGIME + MACRO + PHASE
//────────────────────────────────────
var label banner = na

if barstate.isnew
    label.delete(banner)

    string dayText =
         inCall ? "CALL DAY" :
         inPut  ? "PUT DAY"  :
                  "NEUTRAL DAY"

    string regimeText =
         isTrendRegime   ? "Regime: Trend"  :
         isGrindRegime   ? "Regime: Grind"  :
         isNeutralRegime ? "Regime: Neutral" :
                           "Regime: N/A"

    string macroText =
         macroBullMetals ? "Macro: Bullish metals" :
         macroBearMetals ? "Macro: Bearish metals" :
                           "Macro: Neutral"

    string phaseText =
         ideaState == ideaStateProbe     ? "Phase: PROBE" :
         ideaState == ideaStateCore      ? "Phase: CORE" :
         ideaState == ideaStateExpansion ? "Phase: EXPANSION" :
                                           "Phase: OFF"

    string biasText =
         dayBias == 1  ? "Bias: CALL" :
         dayBias == -1 ? "Bias: PUT"  :
                         "Bias: NEUTRAL"

    // Build banner text in two parts to avoid long-line string issues
    string bannerPart1 = dayText + " | Conf: " + str.tostring(confidence, "#") + "/5"
    string bannerPart2 = " | " + regimeText + " | " + macroText + " | " + phaseText + " | " + biasText
    string bannerText  = bannerPart1 + bannerPart2

    color bannerColor =
         inCall ? color.new(color.green, 0) :
         inPut  ? color.new(color.red,   0) :
                  color.new(color.gray,  0)

    banner := label.new(
         bar_index, high + atr14 * bannerAtrMult,
         bannerText,
         style     = label.style_label_center,
         color     = bannerColor,
         textcolor = color.white,
         size      = size.large)

//────────────────────────────────────
// PLOTS (VWAP + EMAs)

plot(vwapVal, "VWAP",  color = color.orange, linewidth = 2)
plot(emaFast, "EMA 9", color = color.aqua)
plot(emaSlow, "EMA 21",color = color.blue)

//────────────────────────────────────
// BIG ENTRY / EXIT LABELS + SIZE TEXT
//────────────────────────────────────

// Vertical spacing so labels don’t cover candles
yBuyCall   = low  - atr14 * entryLabelMult
ySellCall  = high + atr14 * exitLabelMult
yBuyPut    = high + atr14 * entryLabelMult
ySellPut   = low  - atr14 * exitLabelMult

// Only show label on *first* bar of each signal streak
firstBuyCall  = buyCall  and not buyCall[1]
firstSellCall = sellCall and not sellCall[1]
firstBuyPut   = buyPut   and not buyPut[1]
firstSellPut  = sellPut  and not sellPut[1]

//────────────────────────────────────
// ENTRY: TRACK RISK, LINES, BARS
//────────────────────────────────────

// Increment bars-in-trade counter
if (inCall or inPut) and not na(barsSinceEntry)
    barsSinceEntry += 1

// Helper: playbook tag
f_setupTag(bool isCall) =>
    isCall ?
         (isTrendRegime ? "Setup: Trend Long" : "Setup: Other Long") :
         (isTrendRegime ? "Setup: Trend Short" : "Setup: Other Short")

// BUY CALL entry block
if firstBuyCall
    // 1R distance in price (vol-adjusted)
    float riskDist = effRiskATRMult * atr14

    // record for R math
    lastEntryPrice   := close
    lastRiskATR      := riskDist
    lastDirection    := +1
    barsSinceEntry   := 0
    tp1Triggered     := false
    tp2Triggered     := false
    timeStopTriggered := false

    // Trend vs scalp RR selection
    float rrLocalCall = isTrendMode ? rrMultTrend : rrMultScalp

    // compute stop/target
    float callStop   = close - riskDist
    float callTarget = close + rrLocalCall * riskDist
    currStopPrice    := callStop
    currTargetPrice  := callTarget

    // new lines starting at entry bar, to be extended until exit
    stopLine   := line.new(bar_index, callStop,   bar_index, callStop,   extend = extend.none, color = color.new(color.red,   0), width = 1)
    targetLine := line.new(bar_index, callTarget, bar_index, callTarget, extend = extend.none, color = color.new(color.green, 0), width = 1)

    // Dynamic $ 1R + contracts (for options, 100 shares)
    float oneR_dollar   = riskDist * 100.0
    float contractsF    = oneR_dollar > 0 ? perTradeRiskDollar / oneR_dollar : 0.0
    int   contractsSize = contractsF > 0 ? int(math.round(contractsF)) : 0

    string setupText = f_setupTag(true)

    // Note grind trend in size logic so you remember to take faster profits
    string sizeBaseCall =
         ideaState == ideaStateOff ? "Size: NONE (campaign off)" :
         isBleedTrend ? "Size: SMALL (GRIND TREND – aim 0.5R–1R)" :
         extendedFromVWAPEntry or (confidence <= 2) or ideaState == ideaStateProbe ? "Size: SMALL (probe / partial)" :
         (not isTrendRegime or macroNeutral or (confidence <= 4) or ideaState == ideaStateCore) ? "Size: NORMAL" :
         (nearVWAPEntry and isTrendRegime and macroBullMetals and confidence == 5 and ideaState == ideaStateExpansion) ?
             "Size: AGGRESSIVE (scale-in only if working)" :
             "Size: NORMAL"

    // Mode label text
    string modeLabelTextCall = isTrendMode ? "\nMode: TREND" : "\nMode: SCALP"

    string helperTextCall = ""
    if showSizeHelper and oneR_dollar > 0
        helperTextCall := "\n" + setupText + "\n1R ≈ $" + str.tostring(oneR_dollar, "#.##") +
                          "\nSize guide: " + str.tostring(contractsSize) + " contracts"

    label.new(
         bar_index, yBuyCall,
         "BUY CALL\n" + sizeBaseCall + modeLabelTextCall + helperTextCall,
         style     = label.style_label_up,
         color     = color.new(color.lime, 0),
         textcolor = color.black,
         size      = size.large)

// BUY PUT entry block
if firstBuyPut
    float riskDist2 = effRiskATRMult * atr14

    lastEntryPrice   := close
    lastRiskATR      := riskDist2
    lastDirection    := -1
    barsSinceEntry   := 0
    tp1Triggered     := false
    tp2Triggered     := false
    timeStopTriggered := false

    float rrLocalPut = isTrendMode ? rrMultTrend : rrMultScalp

    float putStop   = close + riskDist2
    float putTarget = close - rrLocalPut * riskDist2
    currStopPrice   := putStop
    currTargetPrice := putTarget

    stopLine   := line.new(bar_index, putStop,   bar_index, putStop,   extend = extend.none, color = color.new(color.red,   0), width = 1)
    targetLine := line.new(bar_index, putTarget, bar_index, putTarget, extend = extend.none, color = color.new(color.green, 0), width = 1)

    float oneR_dollar2   = riskDist2 * 100.0
    float contractsF2    = oneR_dollar2 > 0 ? perTradeRiskDollar / oneR_dollar2 : 0.0
    int   contractsSize2 = contractsF2 > 0 ? int(math.round(contractsF2)) : 0

    string setupText2 = f_setupTag(false)

    string sizeBasePut =
         ideaState == ideaStateOff ? "Size: NONE (campaign off)" :
         isBleedTrend ? "Size: SMALL (GRIND TREND – aim 0.5R–1R)" :
         extendedFromVWAPEntry or (confidence <= 2) or ideaState == ideaStateProbe ? "Size: SMALL (probe / partial)" :
         (not isTrendRegime or macroNeutral or (confidence <= 4) or ideaState == ideaStateCore) ? "Size: NORMAL" :
         (nearVWAPEntry and isTrendRegime and macroBearMetals and confidence == 5 and ideaState == ideaStateExpansion) ?
             "Size: AGGRESSIVE (scale-in only if working)" :
             "Size: NORMAL"

    string modeLabelTextPut = isTrendMode ? "\nMode: TREND" : "\nMode: SCALP"

    string helperTextPut = ""
    if showSizeHelper and oneR_dollar2 > 0
        helperTextPut := "\n" + setupText2 + "\n1R ≈ $" + str.tostring(oneR_dollar2, "#.##") +
                         "\nSize guide: " + str.tostring(contractsSize2) + " contracts"

    label.new(
         bar_index, yBuyPut,
         "BUY PUT\n" + sizeBasePut + modeLabelTextPut + helperTextPut,
         style     = label.style_label_down,
         color     = color.new(color.red, 0),
         textcolor = color.white,
         size      = size.large)

//────────────────────────────────────
// CURRENT R + STRUCTURED EXIT LABELS
//────────────────────────────────────
float currentR = na
if (inCall or inPut) and not na(lastEntryPrice) and lastRiskATR > 0
    float move       = close - lastEntryPrice
    float signedMove = lastDirection == +1 ? move : -move
    currentR := signedMove / lastRiskATR

// Extend current trade's lines from entry → current/exit bar
if (inCall or inPut or firstSellCall or firstSellPut)
    if not na(stopLine)
        line.set_x2(stopLine, bar_index)
    if not na(targetLine)
        line.set_x2(targetLine, bar_index)

if (inCall or inPut) and not na(currentR)
    // TP1 ~ +1R
    if not tp1Triggered and currentR >= 1.0
        tp1Triggered := true
        label.new(
             bar_index, close,
             "TP1 ~ +1R\nConsider partial",
             style     = (lastDirection == +1 ? label.style_label_down : label.style_label_up),
             color     = color.new(color.yellow, 0),
             textcolor = color.black,
             size      = size.small)

        // Break-even shift on TP1 (optionally +0.2R cushion)
        if not na(lastEntryPrice) and lastRiskATR > 0 and not na(stopLine)
            float beStop = lastDirection == +1 ? lastEntryPrice + 0.2 * lastRiskATR : lastEntryPrice - 0.2 * lastRiskATR
            float newStop = lastDirection == +1 ? math.max(currStopPrice, beStop) : math.min(currStopPrice, beStop)
            if newStop != currStopPrice
                currStopPrice := newStop
                line.set_y1(stopLine, currStopPrice)
                line.set_y2(stopLine, currStopPrice)

    // TP2 ~ +2R
    if not tp2Triggered and currentR >= 2.0
        tp2Triggered := true
        label.new(
             bar_index, close,
             "TP2 ~ +2R\nStart trail / protect",
             style     = (lastDirection == +1 ? label.style_label_down : label.style_label_up),
             color     = color.new(color.orange, 0),
             textcolor = color.black,
             size      = size.small)

    // Simple trailing idea: once TP2 is hit, don't let R go back below +1R
    if tp2Triggered and lastRiskATR > 0 and not na(stopLine)
        float newStopPrice = lastDirection == +1 ?
             math.max(currStopPrice, lastEntryPrice + 1.0 * lastRiskATR) :
             math.min(currStopPrice, lastEntryPrice - 1.0 * lastRiskATR)
        if newStopPrice != currStopPrice
            currStopPrice := newStopPrice
            line.set_y1(stopLine, currStopPrice)
            line.set_y2(stopLine, currStopPrice)

    // Time stop if flat after N bars
    if not timeStopTriggered and not na(barsSinceEntry) and barsSinceEntry >= maxBarsInTrade and math.abs(currentR) < 0.3
        timeStopTriggered := true
        label.new(
             bar_index, close,
             "TIME STOP\nNo progress, consider exit",
             style     = (lastDirection == +1 ? label.style_label_down : label.style_label_up),
             color     = color.new(color.gray, 0),
             textcolor = color.white,
             size      = size.small)

//────────────────────────────────────
// EXIT: SELL CALL with R annotation
//────────────────────────────────────
if firstSellCall
    float tradeR = 0.0
    if not na(lastEntryPrice) and lastRiskATR > 0
        float moveExit       = close - lastEntryPrice
        float signedMoveExit = +1 * moveExit
        tradeR := signedMoveExit / lastRiskATR

    dayPnL_R    += tradeR

    // Push into tradeRs array for stats
    array.push(tradeRs, tradeR)
    if array.size(tradeRs) > statsLookback
        array.shift(tradeRs)

    if ideaState != ideaStateOff
        if dayPnL_R >= 1.0 and ideaState == ideaStateProbe
            ideaState := ideaStateCore
        if dayPnL_R >= 3.0 and ideaState == ideaStateCore
            ideaState := ideaStateExpansion

    lastEntryPrice    := na
    lastRiskATR       := na
    lastDirection     := 0
    tp1Triggered      := false
    tp2Triggered      := false
    timeStopTriggered := false
    barsSinceEntry    := na

    // Keep lines on chart for history; just detach handle.
    stopLine        := na
    targetLine      := na
    currStopPrice   := na
    currTargetPrice := na

    string rTextCall = "R: " + str.tostring(tradeR, "#.##")

    label.new(
         bar_index, ySellCall,
         "SELL CALL\n" + rTextCall,
         style     = label.style_label_down,
         color     = color.new(color.green, 10),
         textcolor = color.black,
         size      = size.large)

// EXIT: SELL PUT with R annotation
if firstSellPut
    float tradeR2 = 0.0
    if not na(lastEntryPrice) and lastRiskATR > 0
        float moveExit2       = lastEntryPrice - close
        float signedMoveExit2 = +1 * moveExit2
        tradeR2 := signedMoveExit2 / lastRiskATR

    dayPnL_R    += tradeR2

    array.push(tradeRs, tradeR2)
    if array.size(tradeRs) > statsLookback
        array.shift(tradeRs)

    if ideaState != ideaStateOff
        if dayPnL_R >= 1.0 and ideaState == ideaStateProbe
            ideaState := ideaStateCore
        if dayPnL_R >= 3.0 and ideaState == ideaStateCore
            ideaState := ideaStateExpansion

    lastEntryPrice    := na
    lastRiskATR       := na
    lastDirection     := 0
    tp1Triggered      := false
    tp2Triggered      := false
    timeStopTriggered := false
    barsSinceEntry    := na

    stopLine        := na
    targetLine      := na
    currStopPrice   := na
    currTargetPrice := na

    string rTextPut = "R: " + str.tostring(tradeR2, "#.##")

    label.new(
         bar_index, ySellPut,
         "SELL PUT\n" + rTextPut,
         style     = label.style_label_up,
         color     = color.new(color.maroon, 10),
         textcolor = color.white,
         size      = size.large)
