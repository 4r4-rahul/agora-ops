        float signedMoveExit2 = +1 * moveExit2
        tradeR2 := signedMoveExit2 / entryRiskATR

    dayPnL_R    += tradeR2
    tradesToday += 1

    if inRTH
        array.push(tradeRsRTH, tradeR2)
        if array.size(tradeRsRTH) > statsLookback
            array.shift(tradeRsRTH)
        todayR_RTH     += tradeR2
        tradesTodayRTH += 1

    if mstrUpgradeActive
        if lastSetupType == "REVERSAL"
            array.push(tradeRsReversal, tradeR2)
            if array.size(tradeRsReversal) > mstrSetupBucketLookback
                array.shift(tradeRsReversal)
        else if lastSetupType == "SCALP"
            array.push(tradeRsScalp, tradeR2)
            if array.size(tradeRsScalp) > mstrSetupBucketLookback
                array.shift(tradeRsScalp)
        else
            array.push(tradeRsTrend, tradeR2)
            if array.size(tradeRsTrend) > mstrSetupBucketLookback
                array.shift(tradeRsTrend)

    if ideaState != ideaStateOff
        if dayPnL_R >= 1.0 and ideaState == ideaStateProbe
            ideaState := ideaStateCore
        if dayPnL_R >= 3.0 and ideaState == ideaStateCore
            ideaState := ideaStateExpansion

    lastEntryPrice    := na
    lastRiskATR       := na
    entryRiskATR      := na
    lastDirection     := 0
    tp1Triggered      := false
    tp2Triggered      := false
    timeStopTriggered := false
    barsSinceEntry    := na
    maxBarsThisTrade  := na
    lastSetupType     := ""

    stopLine        := na
    targetLine      := na
    currStopPrice   := na
    currTargetPrice := na

    string rTextPut = "R: " + str.tostring(tradeR2, "#.##")

    label.new(bar_index, ySellPut, "SELL PUT\n" + rTextPut, style = label.style_label_up, color = color.new(color.maroon, 10), textcolor = color.white, size = size.large)
