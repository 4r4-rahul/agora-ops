//@version=5
indicator("Simple Options Trader (Actually Trades)", overlay = true, max_labels_count = 500)

//════════════════════════════════════════════════════════════════════
// PHILOSOPHY: Trade frequency + simple rules > perfect complexity
// 3 CORE FILTERS: Trend, Structure, Volume
// SIMPLE EXITS: % targets, stop loss, time stop
//════════════════════════════════════════════════════════════════════

//────────────────────────────────────
// CORE INDICATORS
//────────────────────────────────────
vwapVal = ta.vwap(close)
emaFast = ta.ema(close, 9)
emaSlow = ta.ema(close, 21)
atr14 = ta.atr(14)
rsi14 = ta.rsi(close, 14)
volMA = ta.sma(volume, 20)

//────────────────────────────────────
// ADX (TREND STRENGTH) - ONLY FILTER WE NEED
//────────────────────────────────────
adxLen = 14
trueRange = math.max(math.max(high - low, math.abs(high - close[1])), math.abs(low - close[1]))
dmPlus = high - high[1] > low[1] - low ? math.max(high - high[1], 0) : 0
dmMinus = low[1] - low > high - high[1] ? math.max(low[1] - low, 0) : 0

var float trSmooth = 0.0
var float dmPlusSmooth = 0.0
var float dmMinusSmooth = 0.0

if bar_index == 0
    trSmooth := 0
    dmPlusSmooth := 0
    dmMinusSmooth := 0
else
    trSmooth := trSmooth - (trSmooth / adxLen) + trueRange
    dmPlusSmooth := dmPlusSmooth - (dmPlusSmooth / adxLen) + dmPlus
    dmMinusSmooth := dmMinusSmooth - (dmMinusSmooth / adxLen) + dmMinus

diPlus = trSmooth != 0 ? (dmPlusSmooth / trSmooth) * 100 : 0
diMinus = trSmooth != 0 ? (dmMinusSmooth / trSmooth) * 100 : 0
dx = diPlus + diMinus != 0 ? math.abs(diPlus - diMinus) / (diPlus + diMinus) * 100 : 0
adx = ta.sma(dx, 14)

//────────────────────────────────────
// HTF (HIGHER TIMEFRAME FILTER)
//────────────────────────────────────
htfTF = input.timeframe("3", "HTF timeframe", group="Settings")
htfClose = request.security(syminfo.tickerid, htfTF, close)
htfEMA20 = request.security(syminfo.tickerid, htfTF, ta.ema(close, 20))

htfBullish = htfClose > htfEMA20
htfBearish = htfClose < htfEMA20

//────────────────────────────────────
// TIME FILTERS (OPTIONS CRITICAL)
//────────────────────────────────────
hour_of_day = hour(time)
minute_of_day = minute(time)

// NO ENTRIES after 15:45 (can't exit if wrong - OPTIONS CRITICAL)
tooLate = hour_of_day == 15 and minute_of_day >= 45 or hour_of_day >= 16

//────────────────────────────────────
// SIMPLE ENTRY CONDITIONS
//────────────────────────────────────

// CALL: Price > VWAP + EMA crossed + Volume + ADX > 20 + HTF bullish
buyCallCondition = close > vwapVal and emaFast > emaSlow and volume > volMA and adx > 20 and htfBullish and not tooLate

// PUT: Price < VWAP + EMA crossed + Volume + ADX > 20 + HTF bearish  
buyPutCondition = close < vwapVal and emaFast < emaSlow and volume > volMA and adx > 20 and htfBearish and not tooLate

//────────────────────────────────────
// POSITION MANAGEMENT
//────────────────────────────────────
var bool inCall = false
var bool inPut = false
var float entryPrice = na
var int barsInTrade = 0
var float stopPrice = na
var float targetPrice = na

// Settings
stopPercent = input.float(1.5, "Stop Loss %", minval=0.5, step=0.1, group="Risk Management")
targetPercent = input.float(3.0, "Profit Target %", minval=1.0, step=0.5, group="Risk Management")
maxBars = input.int(10, "Max bars in trade", minval=3, maxval=50, group="Risk Management")

// Reset daily
if ta.change(time("D")) != 0
    inCall := false
    inPut := false
    entryPrice := na
    barsInTrade := 0
    stopPrice := na
    targetPrice := na

// Entry signals
buyCall = buyCallCondition and not inCall and not inPut
buyPut = buyPutCondition and not inPut and not inCall

// SIMPLE EXITS (% based - OPTIONS FRIENDLY)
if inCall
    barsInTrade += 1
    // Exit if hit target OR stop OR time stop OR EMA cross
    sellCall = close >= targetPrice or close <= stopPrice or barsInTrade >= maxBars or emaFast < emaSlow
    if sellCall
        inCall := false
        entryPrice := na
        barsInTrade := 0
        label.new(bar_index, high, "SELL CALL\n" + str.tostring((close - entryPrice) / entryPrice * 100, "#.#") + "%", 
                  style=label.style_label_down, color=color.green, textcolor=color.white, size=size.normal)

if inPut
    barsInTrade += 1
    // Exit if hit target OR stop OR time stop OR EMA cross
    sellPut = close <= targetPrice or close >= stopPrice or barsInTrade >= maxBars or emaFast > emaSlow
    if sellPut
        inPut := false
        entryPrice := na
        barsInTrade := 0
        label.new(bar_index, low, "SELL PUT\n" + str.tostring((entryPrice - close) / entryPrice * 100, "#.#") + "%",
                  style=label.style_label_up, color=color.maroon, textcolor=color.white, size=size.normal)

// Entry execution
if buyCall
    inCall := true
    entryPrice := close
    barsInTrade := 0
    stopPrice := close * (1 - stopPercent / 100)
    targetPrice := close * (1 + targetPercent / 100)
    
    // Strike guidance based on ADX
    string strikeText = adx > 35 ? "ATM" : adx > 25 ? "1 OTM" : "2 OTM"
    string expiryText = hour_of_day < 12 ? "0DTE" : "1DTE"
    
    label.new(bar_index, low, "BUY CALL\n🎯 " + strikeText + " | " + expiryText + "\nADX: " + str.tostring(adx, "#"), 
              style=label.style_label_up, color=color.lime, textcolor=color.black, size=size.large)

if buyPut
    inPut := true
    entryPrice := close
    barsInTrade := 0
    stopPrice := close * (1 + stopPercent / 100)
    targetPrice := close * (1 - targetPercent / 100)
    
    string strikeText = adx > 35 ? "ATM" : adx > 25 ? "1 OTM" : "2 OTM"
    string expiryText = hour_of_day < 12 ? "0DTE" : "1DTE"
    
    label.new(bar_index, high, "BUY PUT\n🎯 " + strikeText + " | " + expiryText + "\nADX: " + str.tostring(adx, "#"),
              style=label.style_label_down, color=color.red, textcolor=color.white, size=size.large)

//────────────────────────────────────
// VISUAL AIDS
//────────────────────────────────────
plot(vwapVal, "VWAP", color=color.orange, linewidth=2)
plot(emaFast, "EMA 9", color=color.aqua)
plot(emaSlow, "EMA 21", color=color.blue)

// Status banner
var label banner = na
if barstate.isnew
    label.delete(banner)
    string status = inCall ? "IN CALL" : inPut ? "IN PUT" : "FLAT"
    string adxStatus = adx > 30 ? "ADX: " + str.tostring(adx, "#") + " Strong" : adx > 20 ? "ADX: " + str.tostring(adx, "#") + " OK" : "ADX: " + str.tostring(adx, "#") + " Weak"
    string htfStatus = htfBullish ? "HTF: Bullish" : htfBearish ? "HTF: Bearish" : "HTF: Neutral"
    banner := label.new(bar_index, high + atr14 * 2, status + " | " + adxStatus + " | " + htfStatus,
                        style=label.style_label_center, color=color.gray, textcolor=color.white, size=size.normal)

//────────────────────────────────────
// ALERTS
//────────────────────────────────────
alertcondition(buyCall, "BUY CALL", "BUY CALL signal on {{ticker}}")
alertcondition(buyPut, "BUY PUT", "BUY PUT signal on {{ticker}}")
