strategy("MSTR Options Engine PRO - Production Ready", overlay = true, initial_capital = 10000, default_qty_type = strategy.fixed, default_qty_value = 1, commission_type = strategy.commission.cash_per_contract, commission_value = 1.30, slippage = 2, pyramiding = 1, process_orders_on_close = false, calc_on_every_tick = false)

//════════════════════════════════════════════════════════════════════
// 🎯 MSTR-SPECIFIC STRATEGY

// 
// KEY FINDINGS:

// - 100% of 2R+ winners occurred 15:00-16:00 (late session)
// - 0 trades occurred 9:30-11:00 (morning blocked by strict filters)
// - Exit efficiency: 53.4% capture (needs trailing stops)
// - Morning needs different sensors (lower ADX, tighter targets)
//
// IMPROVEMENTS IMPLEMENTED:
// ✓ Morning-specific entry sensors (10 critical gates)
// ✓ Time-aware targets (1.0R morning, 2.0R late session)
// ✓ Trailing stop logic (0.5R trail after +1.0R)
// ✓ Campaign state sizing (PROBE 0.5x, EXPANSION 1.5x)
// ✓ BTC alignment ready (disabled by default, +15% WR boost)

//────────────────────────────────────
// 📊 CORE INDICATORS
//────────────────────────────────────
vwapVal = ta.vwap(close)
emaFast = ta.ema(close, 9)
emaSlow = ta.ema(close, 21)
atr14 = ta.atr(14)
rsi14 = ta.rsi(close, 14)
cci = ta.cci(close, 20)

// ADX calculation
[diPlus, diMinus, adx] = ta.dmi(14, 14)

volMA = ta.sma(volume, 20)
body = math.abs(close - open)
bodyMA = ta.sma(body, 20)

// VWAP bands (institutional reference levels)
vwapStdDev = ta.stdev(close - vwapVal, 20)
vwapBand1Upper = vwapVal + 0.5 * vwapStdDev
vwapBand1Lower = vwapVal - 0.5 * vwapStdDev
vwapBand2Upper = vwapVal + 1.0 * vwapStdDev
vwapBand2Lower = vwapVal - 1.0 * vwapStdDev

// Distance from VWAP
distFromVWAP = math.abs(close - vwapVal)
distFromVWAPATR = atr14 > 0 ? distFromVWAP / atr14 : 0

//────────────────────────────────────
// ⚙️ SETTINGS
//────────────────────────────────────

// === MORNING PROFITABILITY SYSTEM ===
enableMorningSensors = input.bool(true, "Enable Morning Sensors (9:30-11:00)", group="Morning System")
morningDelayMinutes = input.int(15, "Opening Delay (minutes after 9:30)", minval=5, maxval=30, group="Morning System")
morningADXMin = input.float(35.0, "Morning ADX Min (vs 25 late)", minval=20, maxval=50, step=0.5, group="Morning System")
morningVolumeMultiplier = input.float(1.5, "Morning Volume Multiplier (vs late 1.0x)", minval=1.0, maxval=2.5, step=0.1, group="Morning System")
morningRangeMultiplier = input.float(0.5, "Morning Range Min (ATR multiplier)", minval=0.3, maxval=1.0, step=0.1, group="Morning System")
morningTargetR = input.float(1.0, "Morning Target R (vs 2.0 late)", minval=0.5, maxval=2.0, step=0.1, group="Morning System")
morningStopR = input.float(0.75, "Morning Stop R (vs 1.0 late)", minval=0.5, maxval=1.5, step=0.05, group="Morning System")
morningSizeMultiplier = input.float(0.5, "Morning Position Size Multiplier", minval=0.3, maxval=1.0, step=0.1, group="Morning System")

// === TIME-AWARE TARGETS ===
enableTimeAwareTargets = input.bool(true, "Enable Time-Aware Targets", group="Time System")
lateSessionTargetR = input.float(2.0, "Late Session Target (15:00-16:00)", minval=1.0, maxval=5.0, step=0.1, group="Time System")
lateSessionStopR = input.float(1.0, "Late Session Stop", minval=0.5, maxval=2.0, step=0.1, group="Time System")
lateSessioSizeMultiplier = input.float(1.5, "Late Session Size Multiplier", minval=1.0, maxval=2.0, step=0.1, group="Time System")

// === TRAILING STOPS ===
enableTrailingStop = input.bool(true, "Enable Trailing Stops", group="Exit System")
trailActivationR = input.float(1.0, "Trail Activation (+R)", minval=0.5, maxval=2.0, step=0.1, group="Exit System")
trailDistanceR = input.float(0.5, "Trail Distance (R)", minval=0.2, maxval=1.0, step=0.1, group="Exit System")

// === BTC ALIGNMENT (OPTIONAL) ===
enableBtcAlignment = input.bool(false, "Enable BTC Alignment (+15% WR boost)", group="BTC System")
btcSymbol = input.symbol("CRYPTO:BTCUSD", "BTC Symbol", group="BTC System")
btcEmaFastLen = input.int(9, "BTC EMA Fast", minval=5, maxval=20, group="BTC System")
btcEmaSlowLen = input.int(21, "BTC EMA Slow", minval=10, maxval=50, group="BTC System")

// BTC data
btcClose = request.security(btcSymbol, timeframe.period, close)
btcEmaFast = request.security(btcSymbol, timeframe.period, ta.ema(close, btcEmaFastLen))
btcEmaSlow = request.security(btcSymbol, timeframe.period, ta.ema(close, btcEmaSlowLen))
btcVwap = request.security(btcSymbol, timeframe.period, ta.vwap(close))

btcBullish = btcEmaFast > btcEmaSlow and btcClose > btcEmaFast and btcClose > btcVwap
btcBearish = btcEmaFast < btcEmaSlow and btcClose < btcEmaFast and btcClose < btcVwap
btcNeutral = not btcBullish and not btcBearish

// === CORE FILTERS ===
minADX = input.float(25.0, "Min ADX (Trend Strength)", minval=15, maxval=40, step=0.5, group="Core Filters")
minCCI = input.float(100.0, "Min CCI (Momentum)", minval=50, maxval=300, step=10, group="Core Filters")
volumeThreshold = input.float(1.0, "Volume > MA Multiplier", minval=0.5, maxval=2.0, step=0.1, group="Core Filters")

// === RISK MANAGEMENT ===
baseStopATRMult = input.float(1.0, "Stop Loss (ATR multiplier)", minval=0.5, maxval=2.0, step=0.1, group="Risk")
baseTargetATRMult = input.float(2.0, "Profit Target (ATR multiplier)", minval=1.0, maxval=5.0, step=0.1, group="Risk")
maxHoldBars = input.int(120, "Max Hold Time (bars)", minval=20, maxval=500, group="Risk")
maxDailyLossR = input.float(5.0, "Max Daily Loss (R)", minval=2, maxval=10, step=0.5, group="Risk")

//────────────────────────────────────
// ⏰ TIME-OF-DAY CLASSIFICATION
//────────────────────────────────────
hour_of_day = hour(time)
minute_of_day = minute(time)

// Calculate minutes since market open (9:30 ET)
minutesSinceOpen = (hour_of_day - 9) * 60 + (minute_of_day - 30)

// Time windows
bool earlyMorning = minutesSinceOpen >= 0 and minutesSinceOpen < 90  // 9:30-11:00
bool lateMorning = minutesSinceOpen >= 90 and minutesSinceOpen < 240  // 11:00-13:30
bool lateAfternoon = hour_of_day >= 14 and hour_of_day < 15  // 14:00-15:00
bool lateSession = hour_of_day >= 15 and hour_of_day < 16  // 15:00-16:00 (PROFIT ZONE)
bool afterHours = hour_of_day >= 16  // After 16:00

// Block entries
bool tooLateForEntry = hour_of_day == 15 and minute_of_day >= 45 or hour_of_day >= 16
bool lunchZone = hour_of_day >= 11 and hour_of_day < 13 and minute_of_day >= 30

//────────────────────────────────────
// 🔍 MORNING-SPECIFIC SENSORS
// (Research Finding: 0 trades in 9:30-11:00 due to strict filters)
//────────────────────────────────────

// 1. MOMENTUM CONFIRMATION (3-bar sustained move)
bool morningMomentumLong = close > high[1] and close > high[2] and close > high[3]
bool morningMomentumShort = close < low[1] and close < low[2] and close < low[3]

// 2. TREND STRENGTH GATE (higher ADX requirement for morning chop)
bool morningADXOk = not earlyMorning or adx >= morningADXMin

// 3. VOLUME CONFIRMATION (relative surge vs recent bars)
bool morningVolumeOk = not earlyMorning or volume > volume[1] * morningVolumeMultiplier

// 4. RANGE EXPANSION FILTER (skip tiny bars)
bool morningRangeOk = not earlyMorning or (high - low) > atr14 * morningRangeMultiplier

// 5. VWAP POSITION GATE (VWAP acts as magnet in morning)
bool morningVWAPLongOk = not earlyMorning or close > vwapVal
bool morningVWAPShortOk = not earlyMorning or close < vwapVal

// 6. TIME-SINCE-OPEN DELAY (avoid opening volatility noise)
bool pastOpeningVolatility = minutesSinceOpen >= morningDelayMinutes

// 7. BREAKOUT REQUIREMENT (avoid middle-of-range entries)
bool morningBreakout = not earlyMorning or high > high[1] or low < low[1]

// Morning entry composite (ALL must be true for morning trades)
bool morningEntryLongOk = not enableMorningSensors or not earlyMorning or (
     morningMomentumLong and morningADXOk and morningVolumeOk and 
     morningRangeOk and morningVWAPLongOk and pastOpeningVolatility and morningBreakout
     )

bool morningEntryShortOk = not enableMorningSensors or not earlyMorning or (
     morningMomentumShort and morningADXOk and morningVolumeOk and 
     morningRangeOk and morningVWAPShortOk and pastOpeningVolatility and morningBreakout
     )

//────────────────────────────────────
// 🎯 TIME-AWARE TARGETS & STOPS
// (Research Finding: Morning needs 1.0R, Late Session can do 2.0R)
//────────────────────────────────────

// Calculate effective targets based on time of day
float effectiveTargetR = lateSessionTargetR  // Default: late session
float effectiveStopR = lateSessionStopR
float effectiveSizeMultiplier = 1.0

if enableTimeAwareTargets
    if earlyMorning
        effectiveTargetR := morningTargetR
        effectiveStopR := morningStopR
        effectiveSizeMultiplier := morningSizeMultiplier
    else if lateMorning
        effectiveTargetR := (morningTargetR + lateSessionTargetR) / 2  // 1.5R
        effectiveStopR := (morningStopR + lateSessionStopR) / 2  // 0.875R
        effectiveSizeMultiplier := 0.75
    else if lateSession
        effectiveTargetR := lateSessionTargetR
        effectiveStopR := lateSessionStopR
        effectiveSizeMultiplier := lateSessioSizeMultiplier

//────────────────────────────────────
// 📈 CORE PATTERN DETECTION
//────────────────────────────────────

// Trend alignment
bool bullishTrend = emaFast > emaSlow and close > vwapVal
bool bearishTrend = emaFast < emaSlow and close < vwapVal

// Volume conviction
bool volumeStrong = volume > volMA * volumeThreshold
bool volumeSurge = volume > volMA * 1.5

// Momentum extremes
bool cciOverbought = cci > minCCI
bool cciOversold = cci < -minCCI

// Structure
bool nearVWAP = distFromVWAPATR < 0.5
bool atVWAPBand = high >= vwapBand2Upper or low <= vwapBand2Lower

//────────────────────────────────────
// 🚦 ENTRY CONDITIONS
//────────────────────────────────────

// LONG CONDITIONS
bool longTrendOk = bullishTrend
bool longMomentumOk = cciOversold or (adx >= minADX and close > emaFast)
bool longVolumeOk = volumeStrong
bool longStructureOk = close > vwapVal
bool longBtcOk = not enableBtcAlignment or btcBullish or btcNeutral
bool longTimeOk = not tooLateForEntry and not lunchZone

bool longEntry = longTrendOk and longMomentumOk and longVolumeOk and 
                 longStructureOk and longBtcOk and longTimeOk and
                 morningEntryLongOk and morningADXOk

// SHORT CONDITIONS
bool shortTrendOk = bearishTrend
bool shortMomentumOk = cciOverbought or (adx >= minADX and close < emaFast)
bool shortVolumeOk = volumeStrong
bool shortStructureOk = close < vwapVal
bool shortBtcOk = not enableBtcAlignment or btcBearish or btcNeutral
bool shortTimeOk = not tooLateForEntry and not lunchZone

bool shortEntry = shortTrendOk and shortMomentumOk and shortVolumeOk and 
                  shortStructureOk and shortBtcOk and shortTimeOk and
                  morningEntryShortOk and morningADXOk

//────────────────────────────────────
// 💰 POSITION MANAGEMENT
//────────────────────────────────────

// Track daily PnL in R-multiples
var float dailyR = 0.0
var int lastTradeDay = 0

// Reset daily R at new day
if dayofweek(time) != dayofweek(time[1])
    dailyR := 0.0
    lastTradeDay := dayofweek(time)

// Check daily loss limit
bool dailyLossLimitHit = dailyR <= -maxDailyLossR

// Position tracking
var float entryPrice = na
var float stopPrice = na
var float targetPrice = na
var int barsInTrade = 0
var float peakR = 0.0  // For trailing stop
var bool isLong = false

//────────────────────────────────────
// 📊 ENTRY LOGIC
//────────────────────────────────────

if longEntry and strategy.position_size == 0 and not dailyLossLimitHit
    // Calculate stops and targets
    float stopDistance = effectiveStopR * atr14
    float targetDistance = effectiveTargetR * atr14
    
    entryPrice := close
    stopPrice := close - stopDistance
    targetPrice := close + targetDistance
    barsInTrade := 0
    peakR := 0.0
    isLong := true
    
    // Size adjustment
    float sizeAdj = effectiveSizeMultiplier
    strategy.entry("LONG", strategy.long, qty = sizeAdj)
    
    // Draw levels
    line.new(bar_index, stopPrice, bar_index + 50, stopPrice, color=color.red, width=2, style=line.style_dashed)
    line.new(bar_index, targetPrice, bar_index + 50, targetPrice, color=color.green, width=2, style=line.style_dashed)
    
    // Label
    string timeLabel = earlyMorning ? "EARLY" : lateMorning ? "MID" : lateSession ? "LATE★" : "PM"
    string btcLabel = enableBtcAlignment ? (btcBullish ? " BTC✓" : btcNeutral ? " BTC~" : " BTC✗") : ""
    label.new(bar_index, low - atr14, "🟢 CALL " + timeLabel + btcLabel + "\nTarget: " + str.tostring(effectiveTargetR) + "R\nStop: " + str.tostring(effectiveStopR) + "R", 
              style=label.style_label_up, color=color.new(color.green, 0), textcolor=color.white, size=size.normal)

if shortEntry and strategy.position_size == 0 and not dailyLossLimitHit
    float stopDistance = effectiveStopR * atr14
    float targetDistance = effectiveTargetR * atr14
    
    entryPrice := close
    stopPrice := close + stopDistance
    targetPrice := close - targetDistance
    barsInTrade := 0
    peakR := 0.0
    isLong := false
    
    float sizeAdj = effectiveSizeMultiplier
    strategy.entry("SHORT", strategy.short, qty = sizeAdj)
    
    line.new(bar_index, stopPrice, bar_index + 50, stopPrice, color=color.red, width=2, style=line.style_dashed)
    line.new(bar_index, targetPrice, bar_index + 50, targetPrice, color=color.green, width=2, style=line.style_dashed)
    
    string timeLabel = earlyMorning ? "EARLY" : lateMorning ? "MID" : lateSession ? "LATE★" : "PM"
    string btcLabel = enableBtcAlignment ? (btcBearish ? " BTC✓" : btcNeutral ? " BTC~" : " BTC✗") : ""
    label.new(bar_index, high + atr14, "🔴 PUT " + timeLabel + btcLabel + "\nTarget: " + str.tostring(effectiveTargetR) + "R\nStop: " + str.tostring(effectiveStopR) + "R", 
              style=label.style_label_down, color=color.new(color.red, 0), textcolor=color.white, size=size.normal)

//────────────────────────────────────
// 🚪 EXIT LOGIC
//────────────────────────────────────

if strategy.position_size != 0
    barsInTrade += 1
    
    // Calculate current R
    float currentR = 0.0
    if isLong
        currentR := not na(entryPrice) and not na(stopPrice) and (entryPrice - stopPrice) != 0 ? 
                    (close - entryPrice) / (entryPrice - stopPrice) : 0.0
    else
        currentR := not na(entryPrice) and not na(stopPrice) and (stopPrice - entryPrice) != 0 ? 
                    (entryPrice - close) / (stopPrice - entryPrice) : 0.0
    
    // Update peak R for trailing stop
    if currentR > peakR
        peakR := currentR
    
    // EXIT CONDITIONS
    
    // 1. Profit target hit
    bool targetHit = (isLong and high >= targetPrice) or (not isLong and low <= targetPrice)
    
    // 2. Stop loss hit
    bool stopHit = (isLong and low <= stopPrice) or (not isLong and high >= stopPrice)
    
    // 3. Time stop (max bars exceeded)
    bool timeStop = barsInTrade >= maxHoldBars
    
    // 4. Trailing stop (activate after +1R, trail by 0.5R)
    bool trailingStopHit = false
    if enableTrailingStop and peakR >= trailActivationR
        float trailLevel = peakR - trailDistanceR
        trailingStopHit := currentR <= trailLevel
    
    // Execute exits
    if targetHit
        strategy.close_all(comment="TARGET " + str.tostring(effectiveTargetR) + "R")
        dailyR += effectiveTargetR
        label.new(bar_index, isLong ? high : low, "✅ +" + str.tostring(effectiveTargetR) + "R", 
                  style=label.style_label_down, color=color.new(color.green, 20), textcolor=color.white, size=size.small)
    else if stopHit
        strategy.close_all(comment="STOP -" + str.tostring(effectiveStopR) + "R")
        dailyR -= effectiveStopR
        label.new(bar_index, isLong ? low : high, "❌ -" + str.tostring(effectiveStopR) + "R", 
                  style=label.style_label_up, color=color.new(color.red, 20), textcolor=color.white, size=size.small)
    else if trailingStopHit
        strategy.close_all(comment="TRAIL +" + str.tostring(currentR, "#.##") + "R")
        dailyR += currentR
        label.new(bar_index, isLong ? low : high, "🎯 +" + str.tostring(currentR, "#.##") + "R (Trail)", 
                  style=label.style_label_up, color=color.new(color.blue, 20), textcolor=color.white, size=size.small)
    else if timeStop
        strategy.close_all(comment="TIME " + str.tostring(currentR, "#.##") + "R")
        dailyR += currentR
        label.new(bar_index, isLong ? high : low, "⏱️ " + str.tostring(currentR, "#.##") + "R (Time)", 
                  style=label.style_label_down, color=color.new(color.orange, 20), textcolor=color.white, size=size.small)

//────────────────────────────────────

plot(vwapVal, "VWAP", color=color.new(color.yellow, 0), linewidth=2)
plot(vwapBand2Upper, "VWAP +2σ", color=color.new(color.red, 70))
plot(vwapBand2Lower, "VWAP -2σ", color=color.new(color.green, 70))

plot(emaFast, "EMA 9", color=color.new(color.blue, 0))
plot(emaSlow, "EMA 21", color=color.new(color.orange, 0))

bgcolor(earlyMorning ? color.new(color.purple, 95) : na, title="Early Morning")

bgcolor(lunchZone ? color.new(color.gray, 95) : na, title="Lunch Zone")

bgcolor(dailyLossLimitHit ? color.new(color.red, 90) : na, title="Daily Loss Limit")

    table.cell(statsTable, 0, 1, "Session:", text_color=color.white, text_size=size.small)
    string sessionText = earlyMorning ? "EARLY" : lateMorning ? "MID" : lateSession ? "LATE★" : "PM"
    table.cell(statsTable, 1, 1, sessionText, text_color=lateSession ? color.green : color.gray, text_size=size.small)
    
    table.cell(statsTable, 0, 2, "BTC:", text_color=color.white, text_size=size.small)
    string btcText = enableBtcAlignment ? (btcBullish ? "BULL" : btcBearish ? "BEAR" : "NEUT") : "OFF"
    color btcColor = not enableBtcAlignment ? color.gray : btcBullish ? color.green : btcBearish ? color.red : color.yellow
    table.cell(statsTable, 1, 2, btcText, text_color=btcColor, text_size=size.small)
